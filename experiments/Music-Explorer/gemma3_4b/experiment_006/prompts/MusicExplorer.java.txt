    public abstract class ItemViewAdapter extends ArrayAdapter<itemModel> {

        private int itemTextViewId = 0;
        private int itemImageViewId = 0;
        private int itemLayoutId = 0;

        public ItemViewAdapter(Context context, ArrayList<itemModel> list,int textViewId,int imageViewId,int layoutId) {
            super(context, 0, list);
            itemTextViewId = textViewId;
            itemImageViewId = imageViewId;
            itemLayoutId = layoutId;
        }

        abstract String cleanName(String theFileName);

        @Override
        public View getView(int position,View convertView,ViewGroup parent) {

            View itemView = convertView;
            if ( null == itemView )
                itemView = LayoutInflater.from(getContext()).inflate(itemLayoutId, parent, false);

            itemModel model = getItem(position);

            model.setImageView(itemView.findViewById(itemImageViewId));
            model.setTextView(itemView.findViewById(itemTextViewId));

            TextView textView = itemView.findViewById(itemTextViewId);

            textView.setText(cleanName(model.getName()));

            itemView.setOnClickListener(new View.OnClickListener() {
                @Override
                public void onClick(View v) {
                    theMusicExplorerInterface.onItemClicked(((itemModel)v.getTag()).getURi(),null,false);
                }
            });

            itemView.setTag(model);

            return itemView;
        }

    }

    public void displayFoldersAndFiles(DocumentFile innerPath) {

        filesAndFolders theEntities = theUtilities.getCurrentFilesAndFolders(innerPath);

        if ( null == theEntities ) {
            properties.resetPreferences();
            startActivity(new Intent(this, Settings.class));
            return;
        }

        FolderViewAdapter foldersViewAdapter = (FolderViewAdapter)foldersGridView.getAdapter();
        if ( ! ( null == foldersViewAdapter ) ) {
            foldersViewAdapter.clear();
            foldersViewAdapter.notifyDataSetChanged();
        }

        theFolderItems.clear();

        TrackViewAdapter tracksViewAdapter = (TrackViewAdapter)tracksGridView.getAdapter();
        if ( ! ( null == tracksViewAdapter ) ) {
            tracksViewAdapter.clear();
            tracksViewAdapter.notifyDataSetChanged();
        }

        theTrackItems.clear();

        int k = 0;
        for ( String folderName : theEntities.getFolderNames() )
            theFolderItems.add(new itemModel(folderName, theEntities.getFolderURis().get(k++)));

        k = 0;
        for ( String fileName : theEntities.getFileNames() )
            theTrackItems.add(new itemModel(fileName,theEntities.getFileURis().get(k++)));

        actionBar.setDisplayHomeAsUpEnabled(null == innerPath || uriToName(innerPath.getUri()).equals(properties.getRootFolderName()) ? false : true);

        if ( null == foldersViewAdapter ) {
            foldersViewAdapter = new FolderViewAdapter(this,theFolderItems);
            foldersGridView.setAdapter(foldersViewAdapter);
        }

        foldersViewAdapter.notifyDataSetChanged();

        if ( null == tracksViewAdapter ) {
            tracksViewAdapter = new TrackViewAdapter(this,theTrackItems);
            tracksGridView.setAdapter(tracksViewAdapter);
        }

        tracksViewAdapter.notifyDataSetChanged();

        layoutNotified = false;

        if ( theMusicExplorerInterface.isRoot() )
            this.setTitle(this.getString(R.string.app_name));
        else
            this.setTitle(properties.getCurrentFolder().getName().toString());

        ((TextView)findViewById(R.id.no_files_view)).setVisibility(0 == theEntities.size() ? View.VISIBLE : View.INVISIBLE);

        homeView.getViewTreeObserver().addOnGlobalLayoutListener(new ViewTreeObserver.OnGlobalLayoutListener() {
            @Override
            public void onGlobalLayout() {
                homeView.getViewTreeObserver().removeOnGlobalLayoutListener(this);
                layoutFoldersAndFiles();
            }
        });

    }

    public void layoutFoldersAndFiles() {

        WindowMetrics windowMetrics = this.getSystemService(WindowManager.class).getCurrentWindowMetrics();
        Rect bounds = windowMetrics.getBounds();

        foldersContainerViewWidth = bounds.width();
        homeViewHeight = bounds.height();

        int columnCount = foldersContainerViewWidth / properties.getFolderViewWidth();

        int rowCount = Math.max(1, theFolderItems.size() / columnCount);
        if ( rowCount * columnCount < theFolderItems.size() )
            rowCount += 1;

        if ( theFolderItems.isEmpty() )
            foldersContainerViewHeight = 0;
        else {
            if ( 0 < theTrackItems.size() )
                foldersContainerViewHeight = Math.min(rowCount * properties.getFolderViewHeight(),2 * homeViewHeight / 3);
            else
                foldersContainerViewHeight = homeViewHeight;
        }

        foldersGridView.setNumColumns(columnCount);
        RelativeLayout.LayoutParams params = new RelativeLayout.LayoutParams(foldersContainerViewWidth, foldersContainerViewHeight);
        foldersGridView.setLayoutParams(params);

        layoutFiles();

        if ( ! layoutNotified )
            Util.broadcast(this,"com.envisionate.musicinfolders.NAVIGATION_NOTIFY");

        layoutNotified = true;

    }

    private void layoutFiles() {

        tracksContainerViewWidth = foldersContainerViewWidth;
        tracksContainerViewHeight = homeViewHeight - foldersContainerViewHeight;

        int columnCount = tracksContainerViewWidth / properties.getFileViewWidth();

        tracksGridView.setNumColumns(columnCount);
        RelativeLayout.LayoutParams params = new RelativeLayout.LayoutParams(tracksContainerViewWidth, tracksContainerViewHeight);
        params.topMargin = foldersContainerViewHeight;
        filesFrameLayout.setLayoutParams(params);

    }

    public filesAndFolders getDisplayedFilesAndFolder() {

        if ( theFolderItems.isEmpty() && theTrackItems.isEmpty() )
            return null;

        ArrayList<String> folders = new ArrayList<String>();
        ArrayList<String> folderURis = new ArrayList<String>();
        ArrayList<String> files = new ArrayList<String>();
        ArrayList<String> fileURis = new ArrayList<String>();

        for ( itemModel im : theFolderItems ) {
            folders.add(im.getName());
            folderURis.add(im.getURi());
        }

        for ( itemModel im : theTrackItems ) {
            files.add(im.getName());
            fileURis.add(im.getURi());
        }

        return new filesAndFolders(folders,files,folderURis,fileURis);
    }

    public String uriToName(Uri theUri) {
        if ( null == theUri )
            return null;
        String [] pathNames = theUri.getPath().split(":");
        if ( 0 < pathNames.length )
            return pathNames[pathNames.length - 1];
        return null;
    }

    public DocumentFile getParentOf(DocumentFile aFolder,Boolean isNavigation) {
        if ( null == aFolder )
            return null;
        String parentUri = dbHelper.sqlQueryString("SELECT parentUri FROM folders WHERE folderUri = ?",new String [] {aFolder.getUri().toString()} );
        if ( null == parentUri )
            return null;
        return DocumentFile.fromTreeUri(this,Uri.parse(parentUri));
    }
